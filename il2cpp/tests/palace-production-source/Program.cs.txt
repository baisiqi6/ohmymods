    private static void CheckReason(bool parsed,string reason,string expected,string name)
    {
        _checks++;
        if(parsed)throw new Exception("accepted: "+name);
        if(expected!=null&&reason!=expected)throw new Exception("FAIL reason: "+name+" expected="+expected+" actual="+reason);
    }

    private static string Sha(string text)=>Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text))).ToLowerInvariant();

    private static int Main(string[] args)
    {
        if(args.Length!=2)
        {
            Console.Error.WriteLine("usage: PalaceProductionSource.Tests.dll <registry.json> <layout.json>");
            return 2;
        }
        _registry=File.ReadAllText(args[0]);
        _layout=File.ReadAllText(args[1]);

        CheckSynthetic();
        CheckMutations();

    private static int DistrictAt(PalaceProductionSource source,double designPixelX)
    {
        for(int i=0;i<source.Districts.Count;i++)
        {
            var district=source.Districts[i];
            if(designPixelX>=district.DesignPixelMinX&&designPixelX<district.DesignPixelMaxX)return district.Id;
            if(i==source.Districts.Count-1&&designPixelX==district.DesignPixelMaxX)return district.Id;
        }
        return 0;
    }

    private static bool SourceCloneFacts()
    {
        using var doc=JsonDocument.Parse(_registry);
        var rows=doc.RootElement.GetProperty("assets").EnumerateArray().ToArray();
        bool Compare(string id)
        {
            JsonElement row=default;
            foreach(var candidate in rows)if(candidate.GetProperty("id").GetString()==id){row=candidate;break;}
            if(row.ValueKind==JsonValueKind.Undefined||!_source.TryGetAsset(id,out var asset))return false;
            bool same=asset.Support.GetRawText()==row.GetProperty("support").GetRawText()
                &&asset.Footprint.GetRawText()==row.GetProperty("footprint").GetRawText()
                &&asset.Interaction.GetRawText()==row.GetProperty("interaction").GetRawText()
                &&asset.MotionGroup.GetRawText()==row.GetProperty("motionGroup").GetRawText()
                &&asset.AmodalStatus.GetRawText()==row.GetProperty("amodalStatus").GetRawText();
            if(row.TryGetProperty("waterFit",out var waterFit))same&=asset.WaterFit.GetRawText()==waterFit.GetRawText();
            if(row.TryGetProperty("provenance",out var provenance))same&=asset.Provenance.GetRawText()==provenance.GetRawText();
            return same;
        }
        return Compare("cargo-granary")&&Compare("quay-vertical-brick-wall")&&Compare("east-celestial-vista")
            &&Compare("rear-palace-twin-wing-court")&&Compare("central-palace-gate")&&Compare("court-dancer")
            &&Compare("street-stone-paving");
    }

    private static void CheckReadOnly()
    {
        bool threw=false;
        try{((IList<PalaceProductionSource.Asset>)_source.Assets).Add(null);}catch(NotSupportedException){threw=true;}
        Check(threw,"assets readonly");
        threw=false;
        try{((IList<PalaceProductionSource.District>)_source.Districts).Add(null);}catch(NotSupportedException){threw=true;}
        Check(threw,"districts readonly");
        threw=false;
        try{((IDictionary<string,int>)_source.RelativeDrawOrderHints).Add("x",1);}catch(NotSupportedException){threw=true;}
        Check(threw,"hints readonly");
    }

    private static void RejectRegistry(Action<JsonObject> mutate,string name)
    {
        var root=RegistryObject();mutate(root);

    }

        Check(parsed,"layout mutation accepted: "+name+": "+reason);
    }

    private static JsonObject Asset(JsonObject root,string id)
    {
        foreach(var node in root["assets"]!.AsArray())
        {
            var row=node!.AsObject();if(row["id"]!.GetValue<string>()==id)return row;
        }
        throw new Exception("missing asset: "+id);
    }

    private static JsonObject Placement(JsonObject root,string id)
    {
        foreach(var node in root["placements"]!.AsArray())
        {
            var row=node!.AsObject();if(row["id"]!.GetValue<string>()==id)return row;
        }
        throw new Exception("missing placement: "+id);
    }
