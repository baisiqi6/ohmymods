// Compilation shell: copies the diagnostic helper region byte-for-byte from one in-memory
// snapshot of the current production file and embeds that snapshot's identity in the test
// assembly. A stale test DLL therefore cannot claim a newer production source.
using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

internal static class Program
{
    private const string BeginMarker = "// <PALACE-PAID-CAPTURE-DIAGNOSTICS-BEGIN>";
    private const string EndMarker = "// <PALACE-PAID-CAPTURE-DIAGNOSTICS-END>";
    private const string WalletPostfixMarker = "private static void WalletPostfix";
    private const string WalletCopyMarker = "CopyValidated(new[]{";
    private const string WalletCopySuffix = "},\"actual-wallet\")";
    private const string ArrivalBeginMarker = "// <PALACE-PAID-ARRIVAL-DIAGNOSTIC-BEGIN>";
    private const string ArrivalEndMarker = "// <PALACE-PAID-ARRIVAL-DIAGNOSTIC-END>";
    private const string HoldMethodBeginMarker = "// <PALACE-PAID-HOLD-METHOD-BEGIN>";
    private const string HoldMethodEndMarker = "// <PALACE-PAID-HOLD-METHOD-END>";
    private const string DefaultGuardBeginMarker = "// <PALACE-PAID-ARRIVAL-DEFAULT-GUARD-BEGIN>";
    private const string DefaultGuardEndMarker = "// <PALACE-PAID-ARRIVAL-DEFAULT-GUARD-END>";
    private const string DefaultHoldBeginMarker = "// <PALACE-PAID-HOLD-DEFAULT-BEGIN>";
    private const string DefaultHoldEndMarker = "// <PALACE-PAID-HOLD-DEFAULT-END>";
    private static readonly UTF8Encoding StrictUtf8 = new UTF8Encoding(false, true);

    private static int Main(string[] args)
    {
        if (args.Length == 3 && args[0] == "legacy") return ExtractLegacy(args[1], args[2]);
        if (args.Length != 2)
            throw new ArgumentException("Usage: Extractor <paid-source.cs> <output-directory>"
                + " | Extractor legacy <baseline-source.cs> <output-directory>");
        string sourcePath = Path.GetFullPath(args[0]);
        string output = Path.GetFullPath(args[1]);
        Directory.CreateDirectory(output);

        byte[] sourceBytes = File.ReadAllBytes(sourcePath);
        string source = StrictUtf8.GetString(sourceBytes);
        string region = ExtractRegion(source);
        string walletExpression = ExtractWalletPostfixExpression(source);
        string arrivalRegion = ExtractMarkedBlock(source, ArrivalBeginMarker, ArrivalEndMarker, "arrival diagnostic");
        string holdMethod = ExtractMarkedBlock(source, HoldMethodBeginMarker, HoldMethodEndMarker, "Hold method");
        string defaultGuard = ExtractDefaultGuard(source);
        string defaultHold = ExtractDefaultHold(source);
        byte[] regionBytes = StrictUtf8.GetBytes(region);
        string sourceSha = Hash(sourceBytes);
        string regionSha = Hash(regionBytes);

        string generatedPath = Path.Combine(output, "PalacePaidCaptureDiagnostics.Extracted.cs");
        string generated = "// <auto-generated>Extracted verbatim from il2cpp/Patch_PalacePaidResponsibility.cs.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable enable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n" +
            "{\n" +
            region + "\n" +
            "}\n";
        byte[] generatedBytes = StrictUtf8.GetBytes(generated);
        File.WriteAllBytes(generatedPath, generatedBytes);
        string generatedSha = Hash(generatedBytes);

        string expressionPath = Path.Combine(output, "WalletPostfixExpression.g.cs");
        string expressionSource = "// <auto-generated>WalletPostfix initializer extracted verbatim; host shim supplies CurrencyMap.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable enable\n" +
            "namespace KingdomEnhancedMod\n" +
            "{\n" +
            "internal static class WalletPostfixExpression\n" +
            "{\n" +
            "    internal static int[] Capture(CurrencyMap c) => new[]{" + walletExpression + "};\n" +
            "}\n" +
            "}\n";
        byte[] expressionBytes = StrictUtf8.GetBytes(expressionSource);
        File.WriteAllBytes(expressionPath, expressionBytes);
        string expressionSha = Hash(expressionBytes);

        string arrivalPath = Path.Combine(output, "ArrivalDiagnostic.Extracted.cs");
        string arrivalGenerated = "// <auto-generated>Arrival guard/Hold methods extracted verbatim from\n" +
            "// il2cpp/Patch_PalacePaidResponsibility.cs; host shim supplies game/handle state.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable enable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n" +
            "{\n" +
            "internal static partial class Engine\n" +
            "{\n" +
            arrivalRegion + "\n" +
            holdMethod + "\n" +
            "    private static bool DefaultGuard09(PalacePaidHandle handle,Game game,string tripId,int targetLand)\n" +
            "        => " + defaultGuard + ";\n" +
            "    private static void DefaultHold09(PalacePaidHandle h,string reason){" + defaultHold + "}\n" +
            "}\n" +
            "}\n";
        byte[] arrivalBytes = StrictUtf8.GetBytes(arrivalGenerated);
        File.WriteAllBytes(arrivalPath, arrivalBytes);
        string arrivalSha = Hash(arrivalBytes);

        string identityPath = Path.Combine(output, "PalacePaidCaptureDiagnostics.Identity.cs");
        string identity = "// <auto-generated>Compiled identity snapshot; do not edit.</auto-generated>\n" +
            "namespace KingdomEnhancedMod\n" +
            "{\n" +
            "internal static class PalacePaidCaptureDiagnosticsIdentity\n" +
            "{\n" +
            "    internal const string SourcePath = " + Quote(sourcePath) + ";\n" +
            "    internal const string SourceSha256 = \"" + sourceSha + "\";\n" +
            "    internal const int SourceBytes = " + sourceBytes.Length + ";\n" +
            "    internal const string RegionSha256 = \"" + regionSha + "\";\n" +
            "    internal const int RegionChars = " + region.Length + ";\n" +
            "    internal const string GeneratedPath = " + Quote(generatedPath) + ";\n" +
            "    internal const string GeneratedSha256 = \"" + generatedSha + "\";\n" +
            "    internal const int GeneratedBytes = " + generatedBytes.Length + ";\n" +
            "    internal const string WalletPostfixExpression = " + Quote(walletExpression) + ";\n" +
            "    internal const string WalletPostfixGeneratedPath = " + Quote(expressionPath) + ";\n" +
            "    internal const string WalletPostfixGeneratedSha256 = \"" + expressionSha + "\";\n" +
            "    internal const int WalletPostfixGeneratedBytes = " + expressionBytes.Length + ";\n" +
            "    internal const string ArrivalRegionSha256 = \"" + Hash(StrictUtf8.GetBytes(arrivalRegion)) + "\";\n" +
            "    internal const int ArrivalRegionChars = " + arrivalRegion.Length + ";\n" +
            "    internal const string ArrivalGeneratedPath = " + Quote(arrivalPath) + ";\n" +
            "    internal const string ArrivalGeneratedSha256 = \"" + arrivalSha + "\";\n" +
            "    internal const int ArrivalGeneratedBytes = " + arrivalBytes.Length + ";\n" +
            "    internal const string HoldMethodSha256 = \"" + Hash(StrictUtf8.GetBytes(holdMethod)) + "\";\n" +
            "    internal const string DefaultGuardExpression = " + Quote(defaultGuard) + ";\n" +
            "    internal const string DefaultHoldStatement = " + Quote(defaultHold) + ";\n" +
            "}\n" +
            "}\n";
        byte[] identityBytes = StrictUtf8.GetBytes(identity);
        File.WriteAllBytes(identityPath, identityBytes);

        var metadata = new
        {
            sourcePath,
            sourceSha256 = sourceSha,
            sourceBytes = sourceBytes.Length,
            regionSha256 = regionSha,
            regionChars = region.Length,
            generatedPath,
            generatedSha256 = generatedSha,
            generatedBytes = generatedBytes.Length,
            walletPostfixExpression = walletExpression,
            walletPostfixPath = expressionPath,
            walletPostfixSha256 = expressionSha,
            walletPostfixBytes = expressionBytes.Length,
            arrivalRegionSha256 = Hash(StrictUtf8.GetBytes(arrivalRegion)),
            arrivalRegionChars = arrivalRegion.Length,
            arrivalPath,
            arrivalSha256 = arrivalSha,
            arrivalBytes = arrivalBytes.Length,
            holdMethodSha256 = Hash(StrictUtf8.GetBytes(holdMethod)),
            holdMethodChars = holdMethod.Length,
            defaultGuardExpression = defaultGuard,
            defaultHoldStatement = defaultHold,
            identityPath,
            identitySha256 = Hash(identityBytes)
        };
        File.WriteAllText(Path.Combine(output, "extraction.json"),
            JsonSerializer.Serialize(metadata, new JsonSerializerOptions { WriteIndented = true }) + "\n",
            StrictUtf8);
        Console.WriteLine("sourceBytes=" + sourceBytes.Length + " sourceSha=" + sourceSha
            + " regionChars=" + region.Length + " regionSha=" + regionSha
            + " generatedBytes=" + generatedBytes.Length + " generatedSha=" + generatedSha
            + " arrivalChars=" + arrivalRegion.Length + " arrivalSha=" + arrivalSha
            + " defaultGuardChars=" + defaultGuard.Length);
        return 0;
    }

    private static string ExtractRegion(string source)
    {
        int begin = source.IndexOf(BeginMarker, StringComparison.Ordinal);
        if (begin < 0 || source.IndexOf(BeginMarker, begin + BeginMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one begin marker.");
        int beginLineEnd = source.IndexOf('\n', begin);
        if (beginLineEnd < 0) throw new InvalidOperationException("Unterminated begin marker line.");
        int regionStart = beginLineEnd + 1;
        int end = source.IndexOf(EndMarker, regionStart, StringComparison.Ordinal);
        if (end < 0 || source.IndexOf(EndMarker, end + EndMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one end marker after begin.");
        int endLineStart = source.LastIndexOf('\n', end);
        if (endLineStart < regionStart) throw new InvalidOperationException("Empty diagnostic region.");
        return source.Substring(regionStart, endLineStart + 1 - regionStart);
    }

    private static string ExtractWalletPostfixExpression(string source)
    {
        int wallet = source.IndexOf(WalletPostfixMarker, StringComparison.Ordinal);
        if (wallet < 0 || source.IndexOf(WalletPostfixMarker, wallet + WalletPostfixMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one WalletPostfix.");
        int copy = source.IndexOf(WalletCopyMarker, wallet, StringComparison.Ordinal);
        if (copy < 0)
            throw new InvalidOperationException("WalletPostfix capture expression missing.");
        int start = copy + WalletCopyMarker.Length;
        int end = source.IndexOf(WalletCopySuffix, start, StringComparison.Ordinal);
        if (end < 0)
            throw new InvalidOperationException("WalletPostfix capture suffix missing.");
        if (source.IndexOf(WalletCopyMarker, end + WalletCopySuffix.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected one WalletPostfix capture expression.");
        string expression = source.Substring(start, end - start);
        if (expression.Length == 0 || expression.IndexOf('\n') >= 0 || expression.IndexOf('\r') >= 0)
            throw new InvalidOperationException("WalletPostfix capture expression is not a single line.");
        return expression;
    }

    private static string ExtractMarkedBlock(string source, string beginMarker, string endMarker, string label)
    {
        int begin = source.IndexOf(beginMarker, StringComparison.Ordinal);
        if (begin < 0 || source.IndexOf(beginMarker, begin + beginMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one " + label + " begin marker.");
        int beginLineEnd = source.IndexOf('\n', begin);
        if (beginLineEnd < 0) throw new InvalidOperationException("Unterminated " + label + " begin marker line.");
        int regionStart = beginLineEnd + 1;
        int end = source.IndexOf(endMarker, regionStart, StringComparison.Ordinal);
        if (end < 0 || source.IndexOf(endMarker, end + endMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one " + label + " end marker.");
        int endLineStart = source.LastIndexOf('\n', end);
        if (endLineStart < regionStart) throw new InvalidOperationException("Empty " + label + " region.");
        return source.Substring(regionStart, endLineStart + 1 - regionStart);
    }

    private static string ExtractDefaultGuard(string source)
    {
        string block = ExtractMarkedBlock(source, DefaultGuardBeginMarker, DefaultGuardEndMarker, "default guard").Trim();
        if (!block.StartsWith("if(", StringComparison.Ordinal))
            throw new InvalidOperationException("Default guard does not start with if(.");
        int open = "if".Length;
        int depth = 0, close = -1;
        for (int i = open + 1; i < block.Length; i++)
        {
            char c = block[i];
            if (c == '(') depth++;
            else if (c == ')')
            {
                if (depth == 0) { close = i; break; }
                depth--;
            }
        }
        if (close < 0) throw new InvalidOperationException("Default guard has no closing parenthesis.");
        string condition = block.Substring(open + 1, close - open - 1).Trim();
        int separators = 0;
        for (int i = 0; i + 1 < condition.Length; i++)
            if (condition[i] == '|' && condition[i + 1] == '|') separators++;
        if (separators != 7)
            throw new InvalidOperationException("Default guard must contain exactly eight leaves, found separators=" + separators);
        return condition;
    }

    private static string ExtractDefaultHold(string source)
    {
        string statement = ExtractMarkedBlock(source, DefaultHoldBeginMarker, DefaultHoldEndMarker, "default Hold").Trim();
        if (!statement.StartsWith("if(", StringComparison.Ordinal) || !statement.EndsWith(";", StringComparison.Ordinal)
            || statement.IndexOf('\n') >= 0 || statement.IndexOf('\r') >= 0)
            throw new InvalidOperationException("Default Hold statement is not one if statement line.");
        return statement;
    }

    private static string Quote(string value) => "\"" + value.Replace("\\", "\\\\").Replace("\"", "\\\"")
        .Replace("\r", "\\r").Replace("\n", "\\n") + "\"";
    private static string Hash(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();

    private static int ExtractLegacy(string sourcePath, string outputDirectory)
    {
        sourcePath = Path.GetFullPath(sourcePath);
        string output = Path.GetFullPath(outputDirectory);
        Directory.CreateDirectory(output);
        byte[] sourceBytes = File.ReadAllBytes(sourcePath);
        string source = StrictUtf8.GetString(sourceBytes);
        string defaultGuard = ExtractDefaultGuard(source);
        string sourceSha = Hash(sourceBytes);

        string legacyPath = Path.Combine(output, "Arrival.Legacy.g.cs");
        string legacySource = "// <auto-generated>Legacy default arrival guard extracted verbatim from the pinned\n" +
            "// baseline source; comparison-only control, never the shipped guard.</auto-generated>\n" +
            "#nullable enable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n{\n" +
            "internal static partial class Engine\n{\n" +
            "    private static bool LegacyDefaultGuardB6(PalacePaidHandle handle,Game game,string tripId,int targetLand)\n" +
            "        => " + defaultGuard + ";\n" +
            "}\n" +
            "internal static class ArrivalLegacyIdentity\n{\n" +
            "    internal const string SourcePath = " + Quote(sourcePath) + ";\n" +
            "    internal const string SourceSha256 = \"" + sourceSha + "\";\n" +
            "    internal const string DefaultGuardExpression = " + Quote(defaultGuard) + ";\n" +
            "}\n" +
            "}\n";
        byte[] legacyBytes = StrictUtf8.GetBytes(legacySource);
        File.WriteAllBytes(legacyPath, legacyBytes);

        var metadata = new
        {
            sourcePath,
            sourceSha256 = sourceSha,
            sourceBytes = sourceBytes.Length,
            defaultGuardExpression = defaultGuard,
            legacyPath,
            legacySha256 = Hash(legacyBytes),
            legacyBytes = legacyBytes.Length
        };
        File.WriteAllText(Path.Combine(output, "arrival-legacy-extraction.json"),
            JsonSerializer.Serialize(metadata, new JsonSerializerOptions { WriteIndented = true }) + "\n",
            StrictUtf8);
        Console.WriteLine("legacy sourceBytes=" + sourceBytes.Length + " sourceSha=" + sourceSha
            + " defaultGuardChars=" + defaultGuard.Length);
        return 0;
    }
}
