    internal sealed class Game
    {
        internal enum State { Playing, Paused, Other }
        private State _state;
        private int _land;
        private IntPtr _pointer;
        internal bool UnityNull, ThrowState, ThrowPointer;
        internal int StateReads, LandReads, PointerReads, GameObjectReads;
        internal static int OperatorCalls;
        internal State state
        {
            get { StateReads++; if (ThrowState) throw new InvalidOperationException("state"); return _state; }
            set { _state = value; }
        }
        internal int currentLand { get { LandReads++; return _land; } set { _land = value; } }
        internal object? gameObject { get { GameObjectReads++; return null; } }
        internal IntPtr Pointer { get { PointerReads++; if(ThrowPointer)throw new InvalidOperationException("pointer"); return _pointer; } set { _pointer = value; } }
        public static bool operator ==(Game? a, Game? b)
        {
            OperatorCalls++;
            bool an = a is null || a.UnityNull;
            bool bn = b is null || b.UnityNull;
            return an == bn;
        }
        public static bool operator !=(Game? a, Game? b) => !(a == b);
        public override bool Equals(object? o) => o is Game g && this == g;
        public override int GetHashCode() => _pointer.GetHashCode();
    }

        private sealed class ProbeList : IReadOnlyList<int>
        {
            private readonly int[] _values;
            private readonly bool _throwOnIndex, _throwOnCount;
            internal int IndexReads;
            internal ProbeList(int[] values, bool throwOnIndex = false, bool throwOnCount = false)
            { _values = values; _throwOnIndex = throwOnIndex; _throwOnCount = throwOnCount; }
            public int Count => _throwOnCount ? throw new InvalidOperationException("count") : _values.Length;
            public int this[int index]
            {
                get
                {
                    IndexReads++;
                    if (_throwOnIndex) throw new InvalidOperationException("getter");
                    return _values[index];
                }
            }
            public IEnumerator<int> GetEnumerator()
            {
                for (int i = 0; i < _values.Length; i++) yield return _values[i];
            }
            System.Collections.IEnumerator System.Collections.IEnumerable.GetEnumerator() => GetEnumerator();
        }

        private static int Occurrences(string text, string value)
        {
            int count = 0, at = 0;
            while ((at = text.IndexOf(value, at, StringComparison.Ordinal)) >= 0) { count++; at += value.Length; }
            return count;
        }

        private static void TestSourceIdentity()
        {
            string source = File.ReadAllText(PalacePaidCaptureDiagnosticsIdentity.SourcePath, Encoding.UTF8);
            byte[] sourceBytes = Encoding.UTF8.GetBytes(source);
            Check("identity-source-present", source.Length > 0);
            Check("identity-source-bytes", sourceBytes.Length == PalacePaidCaptureDiagnosticsIdentity.SourceBytes);
            Check("identity-source-sha", Sha(sourceBytes) == PalacePaidCaptureDiagnosticsIdentity.SourceSha256);
            // The R3 binary embedded the pre-diagnostic source hash; a stale binary must fail the checks above.
            Check("identity-old-binary-negative",
                PalacePaidCaptureDiagnosticsIdentity.SourceSha256 != "47524151471268980798d616037dec4b56c87eae453a60dd32ba205a553cfe53");

            string region;
            try { region = ExtractRegion(source); }
            catch (Exception e) { Check("identity-region-extract", false, e.Message); return; }
            Check("identity-region-chars", region.Length == PalacePaidCaptureDiagnosticsIdentity.RegionChars);
            Check("identity-region-sha", Sha(Encoding.UTF8.GetBytes(region)) == PalacePaidCaptureDiagnosticsIdentity.RegionSha256);
            Check("identity-region-class", Has(region, "internal sealed class PalacePaidCaptureDiagnostics"));
            Check("identity-region-no-mirror", !Has(region, "class Extracted") && !Has(region, "class Mirror"));
            Check("identity-no-touch-reset", !Has(region, "Touch(") && !Has(region, "_hasTrip"));
            Check("identity-immutable-trip", Has(region, "private readonly Guid _trip;"));

            byte[] generated = File.ReadAllBytes(PalacePaidCaptureDiagnosticsIdentity.GeneratedPath);
            Check("identity-generated-bytes", generated.Length == PalacePaidCaptureDiagnosticsIdentity.GeneratedBytes);
            Check("identity-generated-sha", Sha(generated) == PalacePaidCaptureDiagnosticsIdentity.GeneratedSha256);
            Check("identity-generated-verbatim", Encoding.UTF8.GetString(generated).IndexOf(region, StringComparison.Ordinal) >= 0);

            string mutated = source.Replace("MaxLines=32", "MaxLines=33", StringComparison.Ordinal);
            Check("identity-negative-sha", mutated != source && Sha(Encoding.UTF8.GetBytes(mutated))
                != PalacePaidCaptureDiagnosticsIdentity.SourceSha256);
            bool rejected = false;
            try { ExtractRegion(mutated.Replace("// <PALACE-PAID-CAPTURE-DIAGNOSTICS-END>", "", StringComparison.Ordinal)); }
            catch (InvalidOperationException) { rejected = true; }
            Check("identity-negative-markers", rejected);
        }

        private static void TestDefaultBuildOff()
        {
            string source = File.ReadAllText(PalacePaidCaptureDiagnosticsIdentity.SourcePath, Encoding.UTF8);
            int depth = 0, ifs = 0, endifs = 0, outside = 0;
            foreach (string raw in source.Split('\n'))
            {
                string line = raw.TrimStart();
                if (line.StartsWith("#if PALACE_PAID_CAPTURE_DIAGNOSTICS", StringComparison.Ordinal))
                { depth++; ifs++; continue; }
                if (line.StartsWith("#endif", StringComparison.Ordinal))
                { if (depth > 0) depth--; endifs++; continue; }
                if (depth > 0) continue;
                if (Has(line, "PalacePaidCaptureDiagnostics") || Has(line, "_nullCaptureDiagnostics")
                    || Has(line, "CaptureDiagnostic") || Has(line, "DiagBegin") || Has(line, "DiagMarker")
                    || Has(line, "DiagWallet") || Has(line, "DiagEnd") || Has(line, "DiagPrefs")
                    || Has(line, "DiagCurrencyCompare") || Has(line, "CurrencyComparison")
                    || Has(line, "capturedValue") || Has(line, "observedWalletValue")
                    || Has(line, "DiagResolve") || Has(line, "PALACE-PAID-CAPTURE-DIAGNOSTICS-BEGIN")
                    || Has(line, "PALACE-PAID-CAPTURE-DIAGNOSTICS-END")) outside++;
            }
            Check("default-off-balanced", depth == 0 && ifs == endifs && ifs > 0, "ifs=" + ifs + " endifs=" + endifs);
            Check("default-off-zero-observer", outside == 0, "outside=" + outside);
            Check("default-off-no-unity-debug-log", !Has(source, "UnityEngine.Debug.Log"));
        }

        private static void TestInstanceBudgetIndependence()
        {
            var aLines = Lines();
            var bLines = Lines();
            var a = New(Guid.NewGuid(), aLines);
            var b = New(Guid.NewGuid(), bLines);
            MethodInfo emit = typeof(PalacePaidCaptureDiagnostics).GetMethod("Emit",
                BindingFlags.Instance | BindingFlags.NonPublic)!;
            for (int i = 0; i < 40; i++) emit.Invoke(a, new object[] { "probe-" + i });

            string aText = Join(aLines);
            Check("budget-a-32", aLines.Count == 32, "lines=" + aLines.Count);
            Check("budget-a-prefix", aLines.TrueForAll(x => x.StartsWith("[PALACE-PAID-CAPTURE] ", StringComparison.Ordinal)));
            Check("budget-a-no-33", !Has(aText, "probe-32"));

            BeginAccepted(b);
            b.Resolve(true, false, "no-durable-capture");
            int bCount = bLines.Count;
            Check("budget-b-independent", bCount == 2, "lines=" + bCount);
            Check("budget-b-summary", Has(Join(bLines), "resolve=0:no-durable-capture"));

            BeginAccepted(a);
            a.Resolve(true, false, "no-durable-capture");
            Check("budget-a-not-reset", aLines.Count == 32, "lines=" + aLines.Count);
            Check("old-callbacks-keep-b", bLines.Count == bCount && Has(Join(bLines), "summary trip="), "lines=" + bLines.Count);
        }

        private static bool Same(int[] a, int[] b)
        {
            if (a.Length != b.Length) return false;
            for (int i = 0; i < a.Length; i++) if (a[i] != b[i]) return false;
            return true;
        }

        private sealed class ArrivalFixture
        {
            internal PalacePaidHandle Handle = null!;
            internal PalacePaidHandle Current = null!;
            internal Game? LiveGame;
            internal string TripId = "";
            internal int TargetLand;
            internal List<string> Lines = new List<string>();
            internal bool Result;
            internal Exception? Error;
            internal int OperatorCalls;
            internal string FaultText => Handle.FaultText;
        }

        private static ArrivalFixture RunArrival(bool diag, Action<ArrivalFixture> setup)
        {
            var fx = new ArrivalFixture();
            var d = New(Guid.NewGuid(), fx.Lines);
            fx.Handle = new PalacePaidHandle(Guid.NewGuid(), d);
            fx.LiveGame = new Game { state = Game.State.Playing, currentLand = 12, Pointer = (IntPtr)0x2000 };
            fx.Handle.Game = new Game { Pointer = (IntPtr)0x1000 };
            fx.TripId = fx.Handle.TripId.ToString("D");
            fx.TargetLand = 12;
            fx.Current = fx.Handle;
            setup(fx);
            Engine.SetHandle(fx.Current);
            fx.Handle.TripIdReads = 0;
            fx.Handle.GameReads = 0;
            fx.Handle.Fault.LengthReads = 0;
            if (fx.LiveGame != null)
            {
                fx.LiveGame.StateReads = 0;
                fx.LiveGame.LandReads = 0;
                fx.LiveGame.PointerReads = 0;
            }
            Game.OperatorCalls = 0;
            try
            {
                fx.Result = diag
                    ? Engine.InvokeArrival(fx.Handle, fx.LiveGame!, fx.TripId, fx.TargetLand)
                    : Engine.InvokeDefaultGuard(fx.Handle, fx.LiveGame!, fx.TripId, fx.TargetLand);
            }
            catch (Exception e) { fx.Error = e; }
            fx.OperatorCalls = Game.OperatorCalls;
            return fx;
        }

        private static void CheckArrivalPair(string name, string expectedLeaf, Action<ArrivalFixture> setup,
            int faultReads, int tripReads, int gameReadsDefault, int gameReadsDiag, int stateReads, int landReads,
            int pointerReads, int operators)
        {
            var def = RunArrival(false, setup);
            var diag = RunArrival(true, setup);
            Check(name + "-result-parity", def.Result == diag.Result,
                "default=" + def.Result + " diag=" + diag.Result);
            Check(name + "-exception-parity", def.Error?.GetType() == diag.Error?.GetType(),
                "default=" + def.Error?.GetType().Name + " diag=" + diag.Error?.GetType().Name);
            Check(name + "-default-no-log", def.Lines.Count == 0, "lines=" + def.Lines.Count);
            if (expectedLeaf == "pass" || expectedLeaf.Length == 0)
                Check(name + "-diag-no-log", diag.Lines.Count == 0, "lines=" + diag.Lines.Count);
            else
            {
                Check(name + "-diag-one-log", diag.Lines.Count == 1, "lines=" + diag.Lines.Count);
                Check(name + "-diag-leaf", diag.Lines.Count == 1 && Has(diag.Lines[0], "leaf=" + expectedLeaf),
                    diag.Lines.Count == 0 ? "" : diag.Lines[0]);
                Check(name + "-diag-prefix", diag.Lines.Count == 1
                    && diag.Lines[0].StartsWith("[PALACE-PAID-CAPTURE] trip=", StringComparison.Ordinal),
                    diag.Lines.Count == 0 ? "" : diag.Lines[0]);
            }
            Check(name + "-fault-reads", def.Handle.Fault.LengthReads == faultReads
                && diag.Handle.Fault.LengthReads == faultReads,
                "default=" + def.Handle.Fault.LengthReads + " diag=" + diag.Handle.Fault.LengthReads);
            Check(name + "-trip-reads", def.Handle.TripIdReads == tripReads
                && diag.Handle.TripIdReads == tripReads,
                "default=" + def.Handle.TripIdReads + " diag=" + diag.Handle.TripIdReads);
            Check(name + "-game-reads", def.Handle.GameReads == gameReadsDefault
                && diag.Handle.GameReads == gameReadsDiag,
                "default=" + def.Handle.GameReads + " diag=" + diag.Handle.GameReads);
            Check(name + "-state-reads", def.LiveGame == null
                    ? stateReads == 0 && diag.LiveGame == null
                    : def.LiveGame.StateReads == stateReads && diag.LiveGame!.StateReads == stateReads,
                "default=" + (def.LiveGame == null ? 0 : def.LiveGame.StateReads)
                + " diag=" + (diag.LiveGame == null ? 0 : diag.LiveGame.StateReads));
            Check(name + "-land-reads", def.LiveGame == null
                    ? landReads == 0 && diag.LiveGame == null
                    : def.LiveGame.LandReads == landReads && diag.LiveGame!.LandReads == landReads,
                "default=" + (def.LiveGame == null ? 0 : def.LiveGame.LandReads)
                + " diag=" + (diag.LiveGame == null ? 0 : diag.LiveGame.LandReads));
            Check(name + "-pointer-reads", def.LiveGame == null
                    ? pointerReads == 0 && diag.LiveGame == null
                    : def.LiveGame.PointerReads == pointerReads && diag.LiveGame!.PointerReads == pointerReads,
                "default=" + (def.LiveGame == null ? 0 : def.LiveGame.PointerReads)
                + " diag=" + (diag.LiveGame == null ? 0 : diag.LiveGame.PointerReads));
            Check(name + "-operator-calls", def.OperatorCalls == operators && diag.OperatorCalls == operators,
                "default=" + def.OperatorCalls + " diag=" + diag.OperatorCalls + " expected=" + operators);
        }

                && generatedText.IndexOf(hold, StringComparison.Ordinal) >= 0);

            string expectedGuard = PalacePaidCaptureDiagnosticsIdentity.DefaultGuardExpression;
            Check("arrival-default-exact", expectedGuard
                == "handle!=_handle||handle.Fault.Length!=0||tripId!=handle.TripId.ToString(\"D\")||game==null\n                ||game.state!=Game.State.Playing||game.currentLand!=targetLand||ReferenceEquals(handle.Game,null)||game.Pointer==handle.Game!.Pointer",
                expectedGuard);
            Check("arrival-default-seven-separators", Occurrences(expectedGuard, "||") == 7);
            string[] leaves = {
                "handle!=_handle", "handle.Fault.Length!=0", "tripId!=handle.TripId.ToString(\"D\")", "game==null",
                "game.state!=Game.State.Playing", "game.currentLand!=targetLand", "ReferenceEquals(handle.Game,null)",
                "game.Pointer==handle.Game!.Pointer"
            };
            int last = -1;
            bool ordered = true;
            foreach (string leaf in leaves)
            {
                int at = expectedGuard.IndexOf(leaf, StringComparison.Ordinal);
                if (at < 0 || at <= last) ordered = false;
                last = at;
            }
            Check("arrival-default-eight-order", ordered);
            Check("arrival-default-no-diag", !Has(expectedGuard, "PalacePaidCaptureDiagnostics")
                && !Has(expectedGuard, "Diag"));
            Check("arrival-default-hold-exact",
                PalacePaidCaptureDiagnosticsIdentity.DefaultHoldStatement == "if(h!=null&&h.Fault.Length==0)h.Fault=reason;",
                PalacePaidCaptureDiagnosticsIdentity.DefaultHoldStatement);

            Check("arrival-fault-once", Occurrences(arrival, "handle.Fault.Length") == 1);
            Check("arrival-trip-once", Occurrences(arrival, "handle.TripId.ToString(\"D\")") == 1);
            Check("arrival-live-null-once", Occurrences(arrival, "if(game==null)") == 1);
            Check("arrival-state-once", Occurrences(arrival, "game.state") == 1);
            Check("arrival-land-once", Occurrences(arrival, "game.currentLand") == 1);
            Check("arrival-historical-null-once", Occurrences(arrival, "ReferenceEquals(handle.Game,null)") == 1
                && Occurrences(arrival, "handle.Game==null") == 0);
            Check("arrival-pointer-once", Occurrences(arrival, "game.Pointer==handle.Game!.Pointer") == 1);
            // Pointer identity queries can throw in principle (handle release / native folding);
            // the shipped ObserveExactArrival must keep them fail-closed through catch+Hold.
            Check("arrival-catch-hold-failclosed", Occurrences(source,
                "catch(Exception e){reason=\"arrival:\"+e.GetType().Name;Hold(handle,reason);return false;}") == 1);
            Check("arrival-early-no-fault-read", !Has(arrival.Substring(0,
                arrival.IndexOf("int faultLength=handle.Fault.Length", StringComparison.Ordinal)), "handle.Fault"));
            string region = ExtractRegion(source);
            Check("arrival-class-scalar-only", Has(region, "internal void ArrivalGuard(")
                && Has(region, "internal void FirstFault(") && !Has(region, "UnityEngine") && !Has(region, "Il2Cpp"));
        }

        private static string ExtractMarked(string source, string beginMarker, string endMarker)
        {
            int begin = source.IndexOf(beginMarker, StringComparison.Ordinal);
            if (begin < 0 || source.IndexOf(beginMarker, begin + beginMarker.Length, StringComparison.Ordinal) >= 0)
                throw new InvalidOperationException("begin marker");
            int beginLineEnd = source.IndexOf('\n', begin);
            if (beginLineEnd < 0) throw new InvalidOperationException("begin line");
            int regionStart = beginLineEnd + 1;
            int end = source.IndexOf(endMarker, regionStart, StringComparison.Ordinal);
            if (end < 0 || source.IndexOf(endMarker, end + endMarker.Length, StringComparison.Ordinal) >= 0)
                throw new InvalidOperationException("end marker");
            int endLineStart = source.LastIndexOf('\n', end);
            if (endLineStart < regionStart) throw new InvalidOperationException("empty region");
            return source.Substring(regionStart, endLineStart + 1 - regionStart);
        }

        private static string ExtractRegion(string source)
        {
            const string beginMarker = "// <PALACE-PAID-CAPTURE-DIAGNOSTICS-BEGIN>";
            const string endMarker = "// <PALACE-PAID-CAPTURE-DIAGNOSTICS-END>";
            int begin = source.IndexOf(beginMarker, StringComparison.Ordinal);
            if (begin < 0 || source.IndexOf(beginMarker, begin + beginMarker.Length, StringComparison.Ordinal) >= 0)
                throw new InvalidOperationException("begin marker");
            int beginLineEnd = source.IndexOf('\n', begin);
            if (beginLineEnd < 0) throw new InvalidOperationException("begin line");
            int regionStart = beginLineEnd + 1;
            int end = source.IndexOf(endMarker, regionStart, StringComparison.Ordinal);
            if (end < 0 || source.IndexOf(endMarker, end + endMarker.Length, StringComparison.Ordinal) >= 0)
                throw new InvalidOperationException("end marker");
            int endLineStart = source.LastIndexOf('\n', end);
            if (endLineStart < regionStart) throw new InvalidOperationException("empty region");
            return source.Substring(regionStart, endLineStart + 1 - regionStart);
        }

        private static string Sha(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
