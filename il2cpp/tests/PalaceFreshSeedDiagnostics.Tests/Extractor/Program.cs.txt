// Compilation shell: extracts the real TrySeedVerifiedFresh region and the bounded
// fresh-seed diagnostics helper byte-for-byte from the current production file.
// A stale test DLL cannot claim a newer production source: the generated identity
// carries the source SHA and the test re-reads the source at runtime.
using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

internal static class Program
{
    private const string MethodBegin = "// <PALACE-FRESH-SEED-METHOD-BEGIN>";
    private const string MethodEnd = "// <PALACE-FRESH-SEED-METHOD-END>";
    private const string HelperBegin = "// <PALACE-FRESH-SEED-DIAGNOSTICS-BEGIN>";
    private const string HelperEnd = "// <PALACE-FRESH-SEED-DIAGNOSTICS-END>";
    private static readonly UTF8Encoding StrictUtf8 = new UTF8Encoding(false, true);

    private static int Main(string[] args)
    {
        if (args.Length == 3 && args[0] == "legacy") return ExtractLegacy(args[1], args[2]);
        if (args.Length != 2)
            throw new ArgumentException("Usage: Extractor <visit-clock-source.cs> <output-directory>"
                + " | Extractor legacy <baseline-source.cs> <output-directory>");
        string sourcePath = Path.GetFullPath(args[0]);
        string output = Path.GetFullPath(args[1]);
        Directory.CreateDirectory(output);

        byte[] sourceBytes = File.ReadAllBytes(sourcePath);
        string source = StrictUtf8.GetString(sourceBytes);
        string method = ExtractRegion(source, MethodBegin, MethodEnd);
        string helper = ExtractRegion(source, HelperBegin, HelperEnd);
        string sourceSha = Hash(sourceBytes);
        string methodSha = Hash(StrictUtf8.GetBytes(method));
        string helperSha = Hash(StrictUtf8.GetBytes(helper));

        string defaultPath = Path.Combine(output, "FreshSeed.Default.g.cs");
        string defaultSource = "#undef PALACE_PAID_CAPTURE_DIAGNOSTICS\n" +
            "// <auto-generated>FreshSeed method extracted verbatim from il2cpp/Patch_PalaceVisitClock.cs.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable disable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n{\n" +
            "internal static class FreshSeedDefaultHost\n{\n" +
            method +
            "    internal static readonly object Sync = new object();\n" +
            "    internal static PalaceClockPolicyAdapter Policy = new PalaceClockPolicyAdapter();\n" +
            "    internal static bool HooksReady;\n" +
            "    internal static PalaceScopeObservation Observe(Director instance, out string fail)\n" +
            "    {\n" +
            "        FreshSeedProbe.Bump(ProbeCounter.Observe);\n" +
            "        fail = null;\n" +
            "        return FreshSeedProbe.Observation;\n" +
            "    }\n" +
            "}\n}\n";
        byte[] defaultBytes = StrictUtf8.GetBytes(defaultSource);
        File.WriteAllBytes(defaultPath, defaultBytes);
        string defaultSha = Hash(defaultBytes);

        string diagnosticPath = Path.Combine(output, "FreshSeed.Diagnostic.g.cs");
        string diagnosticSource = "#define PALACE_PAID_CAPTURE_DIAGNOSTICS\n" +
            "// <auto-generated>FreshSeed method and bounded helper extracted verbatim from il2cpp/Patch_PalaceVisitClock.cs.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable disable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n{\n" +
            "internal static class FreshSeedDiagnosticHost\n{\n" +
            method +
            "    internal static readonly object Sync = new object();\n" +
            "    internal static PalaceClockPolicyAdapter Policy = new PalaceClockPolicyAdapter();\n" +
            "    internal static bool HooksReady;\n" +
            "    internal static PalaceScopeObservation Observe(Director instance, out string fail)\n" +
            "    {\n" +
            "        FreshSeedProbe.Bump(ProbeCounter.Observe);\n" +
            "        fail = null;\n" +
            "        return FreshSeedProbe.Observation;\n" +
            "    }\n" +
            "}\n" +
            helper +
            "}\n";
        byte[] diagnosticBytes = StrictUtf8.GetBytes(diagnosticSource);
        File.WriteAllBytes(diagnosticPath, diagnosticBytes);
        string diagnosticSha = Hash(diagnosticBytes);

        string identityPath = Path.Combine(output, "FreshSeed.Identity.cs");
        string identity = "// <auto-generated>Compiled identity snapshot; do not edit.</auto-generated>\n" +
            "namespace KingdomEnhancedMod\n{\n" +
            "internal static class FreshSeedIdentity\n{\n" +
            "    internal const string SourcePath = " + Quote(sourcePath) + ";\n" +
            "    internal const string SourceSha256 = \"" + sourceSha + "\";\n" +
            "    internal const int SourceBytes = " + sourceBytes.Length + ";\n" +
            "    internal const string MethodSha256 = \"" + methodSha + "\";\n" +
            "    internal const int MethodChars = " + method.Length + ";\n" +
            "    internal const string HelperSha256 = \"" + helperSha + "\";\n" +
            "    internal const int HelperChars = " + helper.Length + ";\n" +
            "    internal const string DefaultPath = " + Quote(defaultPath) + ";\n" +
            "    internal const string DefaultSha256 = \"" + defaultSha + "\";\n" +
            "    internal const int DefaultBytes = " + defaultBytes.Length + ";\n" +
            "    internal const string DiagnosticPath = " + Quote(diagnosticPath) + ";\n" +
            "    internal const string DiagnosticSha256 = \"" + diagnosticSha + "\";\n" +
            "    internal const int DiagnosticBytes = " + diagnosticBytes.Length + ";\n" +
            "}\n" +
            "}\n";
        byte[] identityBytes = StrictUtf8.GetBytes(identity);
        File.WriteAllBytes(identityPath, identityBytes);

        var metadata = new
        {
            sourcePath,
            sourceSha256 = sourceSha,
            sourceBytes = sourceBytes.Length,
            methodSha256 = methodSha,
            methodChars = method.Length,
            helperSha256 = helperSha,
            helperChars = helper.Length,
            defaultPath,
            defaultSha256 = defaultSha,
            defaultBytes = defaultBytes.Length,
            diagnosticPath,
            diagnosticSha256 = diagnosticSha,
            diagnosticBytes = diagnosticBytes.Length,
            identityPath,
            identitySha256 = Hash(identityBytes)
        };
        File.WriteAllText(Path.Combine(output, "fresh-seed-extraction.json"),
            JsonSerializer.Serialize(metadata, new JsonSerializerOptions { WriteIndented = true }) + "\n",
            StrictUtf8);
        Console.WriteLine("sourceBytes=" + sourceBytes.Length + " sourceSha=" + sourceSha
            + " methodChars=" + method.Length + " methodSha=" + methodSha
            + " helperChars=" + helper.Length + " helperSha=" + helperSha);
        return 0;
    }

    private static int ExtractLegacy(string sourcePath, string outputDirectory)
    {
        sourcePath = Path.GetFullPath(sourcePath);
        string output = Path.GetFullPath(outputDirectory);
        Directory.CreateDirectory(output);
        byte[] sourceBytes = File.ReadAllBytes(sourcePath);
        string source = StrictUtf8.GetString(sourceBytes);
        string method = ExtractRegion(source, MethodBegin, MethodEnd);
        string sourceSha = Hash(sourceBytes);
        string methodSha = Hash(StrictUtf8.GetBytes(method));

        string legacyPath = Path.Combine(output, "FreshSeed.Legacy.g.cs");
        string legacySource = "#undef PALACE_PAID_CAPTURE_DIAGNOSTICS\n" +
            "// <auto-generated>Legacy TrySeedVerifiedFresh extracted verbatim from the pinned baseline source.\n" +
            "// Do not edit; regenerate through the test project build.</auto-generated>\n" +
            "#nullable disable\n" +
            "using System;\n\n" +
            "namespace KingdomEnhancedMod\n{\n" +
            "internal static class FreshSeedLegacyHost\n{\n" +
            method +
            "    internal static readonly object Sync = new object();\n" +
            "    internal static PalaceClockPolicyAdapter Policy = new PalaceClockPolicyAdapter();\n" +
            "    internal static bool HooksReady;\n" +
            "    internal static PalaceScopeObservation Observe(Director instance, out string fail)\n" +
            "    {\n" +
            "        FreshSeedProbe.Bump(ProbeCounter.Observe);\n" +
            "        fail = null;\n" +
            "        return FreshSeedProbe.Observation;\n" +
            "    }\n" +
            "}\n" +
            "internal static class FreshSeedLegacyIdentity\n{\n" +
            "    internal const string SourceSha256 = \"" + sourceSha + "\";\n" +
            "    internal const string MethodSha256 = \"" + methodSha + "\";\n" +
            "    internal const int MethodChars = " + method.Length + ";\n" +
            "}\n" +
            "}\n";
        byte[] legacyBytes = StrictUtf8.GetBytes(legacySource);
        File.WriteAllBytes(legacyPath, legacyBytes);

        var metadata = new
        {
            sourcePath,
            sourceSha256 = sourceSha,
            sourceBytes = sourceBytes.Length,
            methodSha256 = methodSha,
            methodChars = method.Length,
            legacyPath,
            legacySha256 = Hash(legacyBytes),
            legacyBytes = legacyBytes.Length
        };
        File.WriteAllText(Path.Combine(output, "fresh-seed-legacy-extraction.json"),
            JsonSerializer.Serialize(metadata, new JsonSerializerOptions { WriteIndented = true }) + "\n",
            StrictUtf8);
        Console.WriteLine("legacy sourceBytes=" + sourceBytes.Length + " sourceSha=" + sourceSha
            + " methodChars=" + method.Length + " methodSha=" + methodSha);
        return 0;
    }

    private static string ExtractRegion(string source, string beginMarker, string endMarker)
    {
        int begin = source.IndexOf(beginMarker, StringComparison.Ordinal);
        if (begin < 0 || source.IndexOf(beginMarker, begin + beginMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one " + beginMarker);
        int beginLineEnd = source.IndexOf('\n', begin);
        if (beginLineEnd < 0) throw new InvalidOperationException("Unterminated begin marker line.");
        int regionStart = beginLineEnd + 1;
        int end = source.IndexOf(endMarker, regionStart, StringComparison.Ordinal);
        if (end < 0 || source.IndexOf(endMarker, end + endMarker.Length, StringComparison.Ordinal) >= 0)
            throw new InvalidOperationException("Expected exactly one " + endMarker + " after begin.");
        int endLineStart = source.LastIndexOf('\n', end);
        if (endLineStart < regionStart) throw new InvalidOperationException("Empty extracted region.");
        return source.Substring(regionStart, endLineStart + 1 - regionStart);
    }

    private static string Quote(string value) => "\"" + value.Replace("\\", "\\\\").Replace("\"", "\\\"") + "\"";
    private static string Hash(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
}
