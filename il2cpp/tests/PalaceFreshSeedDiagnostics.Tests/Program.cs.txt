    internal enum ProbeCounter
    {
        ManagersInst, CampaignCurrent, ManagersGame, ManagersDirector,
        GamePointer, GameLand, CampaignLand, CampaignIsland,

    internal static class FreshSeedProbe
    {
        private static readonly int[] Counts = new int[(int)ProbeCounter.Count];
        internal static PalaceScopeObservation Observation = new PalaceScopeObservation();

        internal static void Bump(ProbeCounter counter) => Counts[(int)counter]++;
        internal static void ResetCounts() => Array.Clear(Counts, 0, Counts.Length);
        internal static int[] Snapshot() => (int[])Counts.Clone();
    }

    internal sealed class ProbeList : IReadOnlyList<int>
    {
        private readonly int _count;
        private readonly ListKind _kind;

        internal ProbeList(int count, ListKind kind)
        {
            _count = count;
            _kind = kind;
        }

        public int Count
        {
            get
            {
                FreshSeedProbe.Bump(_kind == ListKind.Objects ? ProbeCounter.ObjectsCount : ProbeCounter.RevisionsCount);
                return _count;
            }
        }

        public int this[int index] => 0;
        public IEnumerator<int> GetEnumerator() { yield break; }
        IEnumerator IEnumerable.GetEnumerator() => GetEnumerator();
    }

            s.ExpectPost = false;

        private static bool RunDefault(Scenario s, out string reason)
        {
            Managers.Inst = s.Managers;
            CampaignSaveData.current = s.Campaign;
            FreshSeedDefaultHost.HooksReady = s.Hooks;
            FreshSeedDefaultHost.Policy.TrialUnknown = s.Trial;
            FreshSeedDefaultHost.Policy.Decision = s.LastDefaultDecision = s.DecisionFactory();
            FreshSeedProbe.ResetCounts();
            return FreshSeedDefaultHost.TrySeedVerifiedFresh(s.Game, s.Island, out reason);
        }

        private static bool RunDiagnostic(Scenario s, out string reason)
        {
            Managers.Inst = s.Managers;
            CampaignSaveData.current = s.Campaign;
            FreshSeedDiagnosticHost.HooksReady = s.Hooks;
            FreshSeedDiagnosticHost.Policy.TrialUnknown = s.Trial;
            FreshSeedDiagnosticHost.Policy.Decision = s.LastDiagnosticDecision = s.DecisionFactory();
            FreshSeedProbe.ResetCounts();
            return FreshSeedDiagnosticHost.TrySeedVerifiedFresh(s.Game, s.Island, out reason);
        }

        private static void TestScenarioMatrix()
        {
            List<Scenario> scenarios = Scenarios();
            Check("scenario-count", scenarios.Count == 32, "count=" + scenarios.Count);
            foreach (Scenario s in scenarios) RunAndCompare(s);
        }

        private static void RunAndCompare(Scenario s)
        {
            bool defaultResult = RunDefault(s, out string defaultReason);
            int[] defaultCounts = FreshSeedProbe.Snapshot();
            int defaultAdopt = s.LastDefaultDecision?.Lease?.Policy.AdoptCalls ?? 0;

            bool diagnosticResult = RunDiagnostic(s, out string diagnosticReason);
            int[] diagnosticCounts = FreshSeedProbe.Snapshot();
            int diagnosticAdopt = s.LastDiagnosticDecision?.Lease?.Policy.AdoptCalls ?? 0;

            Check(s.Name + "-default-result", defaultResult == s.ExpectResult, "got=" + defaultResult);
            Check(s.Name + "-diagnostic-result", diagnosticResult == s.ExpectResult, "got=" + diagnosticResult);
            Check(s.Name + "-counts-equal", Same(defaultCounts, diagnosticCounts),
                Describe(defaultCounts) + " vs " + Describe(diagnosticCounts));
            Check(s.Name + "-adopt-equal", defaultAdopt == diagnosticAdopt, defaultAdopt + " vs " + diagnosticAdopt);
            Check(s.Name + "-adopt-expected",
                diagnosticAdopt == (s.ExpectReason == "complete-native-fresh-facts-day1" ? 1 : 0),
                "got=" + diagnosticAdopt);

            if (s.ExpectLeaf == null)
            {
                Check(s.Name + "-default-reason", defaultReason == s.ExpectReason, defaultReason);
                Check(s.Name + "-diagnostic-reason", diagnosticReason == s.ExpectReason, diagnosticReason);
            }
            else
            {
                Check(s.Name + "-default-pending", defaultReason == "fresh-facts-pending", defaultReason);
                Check(s.Name + "-diagnostic-leaf", diagnosticReason == "fresh-facts-pending:" + s.ExpectLeaf,
                    diagnosticReason);
            }

            int observe = diagnosticCounts[(int)ProbeCounter.Observe];
            int decide = diagnosticCounts[(int)ProbeCounter.Decide];
            Check(s.Name + "-one-observe-one-decide",
                observe == (s.ExpectPost ? 1 : 0) && decide == (s.ExpectPost ? 1 : 0),
                "obs=" + observe + " dec=" + decide);
            Check(s.Name + "-no-extra-native", AllCountsNotGreater(diagnosticCounts, MaxCounts()),
                Describe(diagnosticCounts));
            // fresh 阶段身份不来自 island.land；新方法在任何路径都不得读取该 getter。
            Check(s.Name + "-land-getter-zero", diagnosticCounts[(int)ProbeCounter.IslandLand] == 0,
                Describe(diagnosticCounts));
            Check(s.Name + "-short-circuit", ShortCircuitHolds(s, diagnosticCounts), Describe(diagnosticCounts));
            if (s.ExpectPost)
                Check(s.Name + "-exact-native-counts", Same(diagnosticCounts, MaxCounts()),
                    Describe(diagnosticCounts));
        }

        private static int[] MaxCounts()
        {
            var max = new int[(int)ProbeCounter.Count];
            max[(int)ProbeCounter.ManagersInst] = 1;
            max[(int)ProbeCounter.CampaignCurrent] = 1;
            max[(int)ProbeCounter.ManagersGame] = 2;
            max[(int)ProbeCounter.ManagersDirector] = 2;
            max[(int)ProbeCounter.GamePointer] = 2;
            max[(int)ProbeCounter.GameLand] = 1;
            max[(int)ProbeCounter.CampaignLand] = 1;
            max[(int)ProbeCounter.CampaignIsland] = 2;
            max[(int)ProbeCounter.IslandPointer] = 2;
            max[(int)ProbeCounter.IslandIsNew] = 1;
            max[(int)ProbeCounter.IslandLand] = 0;      // 新 shape 门完全不读存档 land 字段
            max[(int)ProbeCounter.IslandLastReign] = 1;
            max[(int)ProbeCounter.IslandObjects] = 1;

            max[(int)ProbeCounter.Observe] = 1;
            max[(int)ProbeCounter.Decide] = 1;
            return max;
        }

        private static bool ShortCircuitHolds(Scenario s, int[] c)
        {
            switch (s.Name)
            {
                case "hooks-not-ready":
                case "trial-unknown":
                case "game-null":
                case "island-null":
                case "campaign-null":
                case "managers-null":
                    return c[(int)ProbeCounter.ManagersGame] == 0
                        && c[(int)ProbeCounter.CampaignIsland] == 0
                        && c[(int)ProbeCounter.Observe] == 0;
                case "manager-game-null":
                    return c[(int)ProbeCounter.ManagersGame] == 1
                        && c[(int)ProbeCounter.GamePointer] == 0
                        && c[(int)ProbeCounter.ManagersDirector] == 0;
                case "current-game-pointer-mismatch":
                    return c[(int)ProbeCounter.ManagersGame] == 2
                        && c[(int)ProbeCounter.GamePointer] == 2
                        && c[(int)ProbeCounter.ManagersDirector] == 0
                        && c[(int)ProbeCounter.GameLand] == 0;
                case "director-null":
                    return c[(int)ProbeCounter.ManagersDirector] == 1
                        && c[(int)ProbeCounter.GameLand] == 0
                        && c[(int)ProbeCounter.CampaignLand] == 0;
                case "game-land-not-12":
                    return c[(int)ProbeCounter.GameLand] == 1
                        && c[(int)ProbeCounter.CampaignLand] == 0
                        && c[(int)ProbeCounter.CampaignIsland] == 0;
                case "campaign-land-not-12":
                    return c[(int)ProbeCounter.CampaignLand] == 1
                        && c[(int)ProbeCounter.CampaignIsland] == 0
                        && c[(int)ProbeCounter.IslandPointer] == 0;
                case "current-island-null":
                    return c[(int)ProbeCounter.CampaignIsland] == 1
                        && c[(int)ProbeCounter.IslandPointer] == 0
                        && c[(int)ProbeCounter.IslandIsNew] == 0;
                case "current-island-pointer-mismatch":
                    return c[(int)ProbeCounter.CampaignIsland] == 2
                        && c[(int)ProbeCounter.IslandPointer] == 2
                        && c[(int)ProbeCounter.IslandIsNew] == 0;
                case "island-not-new":
                    return c[(int)ProbeCounter.IslandIsNew] == 1
                        && c[(int)ProbeCounter.IslandLastReign] == 0
                        && c[(int)ProbeCounter.IslandObjects] == 0;
                case "lastreign-zero":
                case "lastreign-positive":
                case "lastreign-negative-two":
                    return c[(int)ProbeCounter.IslandLastReign] == 1
                        && c[(int)ProbeCounter.IslandObjects] == 0
                        && c[(int)ProbeCounter.ObjectsCount] == 0;
                case "objects-empty":
                case "objects-nonempty":
                    return c[(int)ProbeCounter.IslandObjects] == 1
                        && c[(int)ProbeCounter.ObjectsCount] == 0
                        && c[(int)ProbeCounter.IslandRevisions] == 0
                        && c[(int)ProbeCounter.RevisionsCount] == 0;
                case "revisions-empty":
                case "revisions-nonempty":
                    return c[(int)ProbeCounter.IslandRevisions] == 1
                        && c[(int)ProbeCounter.RevisionsCount] == 0
                        && c[(int)ProbeCounter.Observe] == 0
                        && c[(int)ProbeCounter.Decide] == 0;
                default:
                    return true;
            }
        }

        private static void TestBoundedDiagnostics()
        {
            ResetFreshSeedDiagnostics();
            List<Scenario> scenarios = Scenarios();
            int initialFailures = 0;
            foreach (Scenario s in scenarios)
            {
                if (s.ExpectLeaf == null || s.ExpectPost) continue;
                RunDiagnostic(s, out _);
                initialFailures++;
            }
            Check("bounded-initial-failures", initialFailures == 21, "count=" + initialFailures);
            Check("bounded-cap-16", PalaceClockLog.Lines.Count == 16, "lines=" + PalaceClockLog.Lines.Count);

            Scenario post = scenarios.Find(x => x.ExpectPost && x.ExpectLeaf != null);
            RunDiagnostic(post, out _);
            Check("bounded-cap-still-16", PalaceClockLog.Lines.Count == 16, "lines=" + PalaceClockLog.Lines.Count);

            ResetFreshSeedDiagnostics();
            Scenario repeat = scenarios.Find(x => x.Name == "hooks-not-ready");
            RunDiagnostic(repeat, out _);
            RunDiagnostic(repeat, out _);
            Check("bounded-dedupe-same", PalaceClockLog.Lines.Count == 1, "lines=" + PalaceClockLog.Lines.Count);

            ResetFreshSeedDiagnostics();
            PalaceClockLog.Throw = true;
            bool survived = true;
            string thrownReason = null;
            try { RunDiagnostic(repeat, out thrownReason); }
            catch (Exception) { survived = false; }
            PalaceClockLog.Throw = false;
            Check("bounded-sink-throw-contained", survived && thrownReason == "fresh-facts-pending:hooks-not-ready",
                "reason=" + thrownReason);
            Check("bounded-sink-no-line-on-throw", PalaceClockLog.Lines.Count == 0,
                "lines=" + PalaceClockLog.Lines.Count);

            ResetFreshSeedDiagnostics();
            Scenario gameNull = scenarios.Find(x => x.Name == "game-null");
            Scenario islandNull = scenarios.Find(x => x.Name == "island-null");
            RunDiagnostic(gameNull, out _);
            RunDiagnostic(gameNull, out _);
            RunDiagnostic(islandNull, out _);
            Check("bounded-transition", PalaceClockLog.Lines.Count == 2, "lines=" + PalaceClockLog.Lines.Count);
            Check("bounded-line-prefix",
                PalaceClockLog.Lines.TrueForAll(x => x.StartsWith("fresh-seed-first-failure fresh-facts-pending:",
                    StringComparison.Ordinal)),
                string.Join("|", PalaceClockLog.Lines));
        }

        private static void TestSeedIdempotenceAndSavedHydrate()
        {
            // 同一 lease 上重复 seed：第二次仍 true，但 AdoptNewIslandDay1 不得再次执行。
            Scenario repeat = Base("repeat-seed");
            var repeatDecision = new PalaceScopeDecision
            {
                Verdict = PalaceScopeVerdict.Owned,
                Lease = new PalaceClockLease()
            };
            repeat.DecisionFactory = () => repeatDecision;
            bool first = RunDiagnostic(repeat, out string firstReason);
            int firstCountsObserve = FreshSeedProbe.Snapshot()[(int)ProbeCounter.Observe];
            bool second = RunDiagnostic(repeat, out string secondReason);
            int[] secondCounts = FreshSeedProbe.Snapshot();
            Check("repeat-seed-first", first && firstReason == "complete-native-fresh-facts-day1", firstReason);
            Check("repeat-seed-second", second && secondReason == "fresh-local-already-established", secondReason);

        private static void TestSourceIdentity()
        {
            string source = File.ReadAllText(FreshSeedIdentity.SourcePath, Encoding.UTF8);
            byte[] sourceBytes = Encoding.UTF8.GetBytes(source);
            Check("identity-source-bytes", sourceBytes.Length == FreshSeedIdentity.SourceBytes);
            Check("identity-source-sha", Sha(sourceBytes) == FreshSeedIdentity.SourceSha256);
            Check("identity-one-method", Occurrences(source, "internal static bool TrySeedVerifiedFresh(") == 1);
            Check("identity-one-method-begin", Occurrences(source, "// <PALACE-FRESH-SEED-METHOD-BEGIN>") == 1);
            Check("identity-one-method-end", Occurrences(source, "// <PALACE-FRESH-SEED-METHOD-END>") == 1);
            Check("identity-one-helper-begin", Occurrences(source, "// <PALACE-FRESH-SEED-DIAGNOSTICS-BEGIN>") == 1);
            Check("identity-one-helper-end", Occurrences(source, "// <PALACE-FRESH-SEED-DIAGNOSTICS-END>") == 1);

            string method = ExtractBetween(source, "// <PALACE-FRESH-SEED-METHOD-BEGIN>",
                "// <PALACE-FRESH-SEED-METHOD-END>");
            string helper = ExtractBetween(source, "// <PALACE-FRESH-SEED-DIAGNOSTICS-BEGIN>",
                "// <PALACE-FRESH-SEED-DIAGNOSTICS-END>");
            Check("identity-method-chars", method.Length == FreshSeedIdentity.MethodChars);
            Check("identity-method-sha", Sha(Encoding.UTF8.GetBytes(method)) == FreshSeedIdentity.MethodSha256);
            Check("identity-helper-chars", helper.Length == FreshSeedIdentity.HelperChars);
            Check("identity-helper-sha", Sha(Encoding.UTF8.GetBytes(helper)) == FreshSeedIdentity.HelperSha256);

            string defaultGenerated = File.ReadAllText(FreshSeedIdentity.DefaultPath, Encoding.UTF8);
            string diagnosticGenerated = File.ReadAllText(FreshSeedIdentity.DiagnosticPath, Encoding.UTF8);
            Check("identity-default-bytes", Encoding.UTF8.GetByteCount(defaultGenerated) == FreshSeedIdentity.DefaultBytes);
            Check("identity-default-sha", Sha(Encoding.UTF8.GetBytes(defaultGenerated)) == FreshSeedIdentity.DefaultSha256);
            Check("identity-diagnostic-bytes",
                Encoding.UTF8.GetByteCount(diagnosticGenerated) == FreshSeedIdentity.DiagnosticBytes);
            Check("identity-diagnostic-sha",
                Sha(Encoding.UTF8.GetBytes(diagnosticGenerated)) == FreshSeedIdentity.DiagnosticSha256);
            Check("identity-default-verbatim-method", defaultGenerated.Contains(method));
            Check("identity-diagnostic-verbatim-method", diagnosticGenerated.Contains(method));
            Check("identity-diagnostic-verbatim-helper", diagnosticGenerated.Contains(helper));
            Check("identity-default-no-helper",
                !defaultGenerated.Contains("internal static class PalaceFreshSeedDiagnostics"));
            Check("identity-default-undef",
                defaultGenerated.StartsWith("#undef PALACE_PAID_CAPTURE_DIAGNOSTICS", StringComparison.Ordinal));
            Check("identity-diagnostic-define",
                diagnosticGenerated.StartsWith("#define PALACE_PAID_CAPTURE_DIAGNOSTICS", StringComparison.Ordinal));

            int ifAt = method.IndexOf("#if PALACE_PAID_CAPTURE_DIAGNOSTICS", StringComparison.Ordinal);
            int elseAt = method.IndexOf("#else", ifAt, StringComparison.Ordinal);
            int endifAt = method.IndexOf("#endif", elseAt, StringComparison.Ordinal);
            Check("identity-gated", ifAt >= 0 && elseAt > ifAt && endifAt > elseAt);
            string defaultBranch = method.Substring(elseAt, endifAt - elseAt);
            Check("identity-default-branch-native-ctor-shape",
                defaultBranch.Contains("!HooksReady || Policy.TrialUnknown || game == null || island == null || c == null")
                && defaultBranch.Contains("|| c.CurrentIsland.Pointer != island.Pointer || !island.isNew")
                && defaultBranch.Contains("|| island.lastPlayedReign != -1 || island.objects != null")
                && defaultBranch.Contains("|| island.savedWithRevisions != null) return false;")
                && !defaultBranch.Contains("island.land")
                && !defaultBranch.Contains(".Count")
                && !defaultBranch.Contains("fresh-facts-pending:"), defaultBranch);
            Check("identity-diagnostic-labels",
                method.Contains("fresh-facts-pending:\" + leaf")
                && method.Contains("fresh-facts-pending:\" + scopeLeaf")
                && method.Contains("leaf = \"lastreign-not-pristine\"")
                && method.Contains("leaf = \"objects-already-initialized\"")
                && method.Contains("leaf = \"revisions-already-initialized\""));
            Check("identity-no-land-read-or-collection-count",
                !method.Contains("island.land ==") && !method.Contains("island.land !=")
                && !method.Contains("objects.Count") && !method.Contains("savedWithRevisions.Count"));
        }

        private static string ExtractBetween(string source, string beginMarker, string endMarker)
        {
            int begin = source.IndexOf(beginMarker, StringComparison.Ordinal);
            if (begin < 0) throw new InvalidOperationException("begin marker");
            int beginLineEnd = source.IndexOf('\n', begin);
            if (beginLineEnd < 0) throw new InvalidOperationException("begin line");
            int regionStart = beginLineEnd + 1;
            int end = source.IndexOf(endMarker, regionStart, StringComparison.Ordinal);
            if (end < 0) throw new InvalidOperationException("end marker");
            int endLineStart = source.LastIndexOf('\n', end);
            if (endLineStart < regionStart) throw new InvalidOperationException("empty region");
            return source.Substring(regionStart, endLineStart + 1 - regionStart);
        }

        private static bool Same(int[] a, int[] b)
        {
            if (a.Length != b.Length) return false;
            for (int i = 0; i < a.Length; i++) if (a[i] != b[i]) return false;
            return true;
        }

        private static bool AllCountsNotGreater(int[] counts, int[] max)
        {
            for (int i = 0; i < counts.Length; i++) if (counts[i] > max[i]) return false;
            return true;
        }

        private static string Describe(int[] counts)
        {
            var parts = new List<string>();
            for (int i = 0; i < counts.Length; i++)
                if (counts[i] != 0) parts.Add(((ProbeCounter)i).ToString() + "=" + counts[i]);
            return string.Join(",", parts);
        }

        private static int Occurrences(string text, string value)
        {
            int count = 0, at = 0;
            while ((at = text.IndexOf(value, at, StringComparison.Ordinal)) >= 0) { count++; at += value.Length; }
            return count;
        }

        private static string Sha(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
    }
