internal static class JsonUtility
{
    internal static string ToJson(object obj, bool prettyPrint)
    {
        IslandSaveData island = obj as IslandSaveData;
        if (island == null) throw new InvalidOperationException("host-only-island");
        if (island.ThrowOnSerialize) throw new InvalidOperationException("snapshot");
        return island.Json;
    }
}

    internal static string Saved(PalaceLayoutRecipe recipe, int? count = null, int? colliderWidth = null,
        bool duplicate = false, bool noGround = false, bool withGround = true, string tilePrefix = "Tile-test-")
    {
        int tiles = count ?? recipe.TileCount;
        int width = colliderWidth ?? recipe.Width;
        var objects = new List<object>();
        if (withGround)
            objects.Add(new
            {
                uniqueID = "Ground-test",
                prefabPath = "Prefabs/Environment/Ground",
                localPosition = new { x = 0, y = 0, z = 0 },
                localScale = new { x = 1, y = 1, z = 1 },
                componentData2 = new[]
                {
                    new { name = "Ground", type = "GroundData",
                        data = JsonSerializer.Serialize(new
                        {
                            colliderOffset = new { x = 0, y = .4375 },
                            colliderSize = new { x = width, y = .875 }
                        }) }
                }
            });
        for (int i = 0; i < tiles; i++)
            objects.Add(new
            {
                uniqueID = tilePrefix + (duplicate ? 0 : i),
                prefabPath = "Prefabs/Environment/Tile Big",
                localPosition = new { x = recipe.Left + 4 * i, y = 0, z = .01 },
                localScale = new { x = 1, y = 1, z = 1 },
                componentData2 = new[]
                {
                    new { name = "Tile", type = "TileData",
                        data = JsonSerializer.Serialize(new { noGround, noSnow = false, snow = new { linkedObjectID = "" } }) }
                }
            });
        return JsonSerializer.Serialize(new { objects });
    }
